toSize(string s) =>
    switch s
        "Tiny"   => size.tiny
        "Small"  => size.small
        "Normal" => size.normal
        "Large"  => size.large
        "Huge"   => size.huge
        => size.auto

parseSymbols(string raw, string exch) =>
    string norm = str.replace_all(raw, "\n", ",")
    norm := str.replace_all(norm, "\r", ",")
    norm := str.replace_all(norm, ";", ",")
    array<string> out = array.new<string>()
    array<string> toks = str.split(norm, ",")
    for i = 0 to array.size(toks) - 1
        string tk = str.trim(array.get(toks, i))
        if tk != "" and not str.startswith(tk, "###")
            array.push(out, str.contains(tk, ":") ? tk : exch + ":" + tk)
    out

fmtPrice(float v) =>
    na(v) ? "–" : str.tostring(v, "#.##")

fmtRsi(float v) =>
    na(v) ? "–" : str.tostring(v, "#.##")

fmtQty(float v) =>
    na(v) or v <= 0 ? "–" : str.tostring(v, "#")

fmtMoney(float v) =>
    na(v) or v <= 0 ? "–" : "₹" + str.tostring(v, "#")

fmtPct(float v) =>
    na(v) ? "–" : str.tostring(v, "#.##") + "%"

fmtDelta(float d, float p) =>
    na(d) or na(p) ? "–" : str.tostring(d, "#.##") + " / " + fmtPct(p)

// ══════════════════════════════════════
// RANKING LOGIC
// Score = Investment Rank + RSI Closeness Rank + SL-Risk Delta % Rank
// Lesser score = better opportunity rank.
// Delta Rank % = selected TF SL% + ((LTP - EMA) / EMA × 100), so SL risk is included.
// ══════════════════════════════════════
array<int> idxArr = array.new<int>()
for i = 0 to array.size(namesArr) - 1
    bool isValid = array.get(validArr, i)
    if isValid or not showQualifiedOnly
        array.push(idxArr, i)

int eligibleCount = array.size(idxArr)
array<int> invRankArr   = array.new<int>(array.size(namesArr), 9999)
array<int> rsiRankArr   = array.new<int>(array.size(namesArr), 9999)
array<int> deltaRankArr = array.new<int>(array.size(namesArr), 9999)
array<float> scoreArr   = array.new<float>(array.size(namesArr), 999999.0)

if eligibleCount > 0
    for a = 0 to eligibleCount - 1
        int ia = array.get(idxArr, a)
        bool va = array.get(validArr, ia)
        if va
            float invA = array.get(investArr, ia)
            float rsiDistA = array.get(rsiArr, ia) - 51.0
            float deltaPctA = array.get(riskDeltaPctArr, ia)
            int invRank = 1
            int rsiRank = 1
            int delRank = 1
            for b = 0 to eligibleCount - 1
                int ib = array.get(idxArr, b)
                bool vb = array.get(validArr, ib)
                if vb and ib != ia
                    float invB = array.get(investArr, ib)
                    float rsiDistB = array.get(rsiArr, ib) - 51.0
                    float deltaPctB = array.get(riskDeltaPctArr, ib)
                    if invB < invA
                        invRank += 1
                    if rsiDistB < rsiDistA
                        rsiRank += 1
                    if deltaPctB < deltaPctA
                        delRank += 1
            array.set(invRankArr, ia, invRank)
            array.set(rsiRankArr, ia, rsiRank)
            array.set(deltaRankArr, ia, delRank)
            array.set(scoreArr, ia, float(invRank + rsiRank + delRank))

array<int> displayIdxArr = array.copy(idxArr)
if eligibleCount > 1
    for a = 0 to eligibleCount - 2
        for b = a + 1 to eligibleCount - 1
            int ia = array.get(displayIdxArr, a)
            int ib = array.get(displayIdxArr, b)
            float scoreA = array.get(scoreArr, ia)
            float scoreB = array.get(scoreArr, ib)
            float investA = nz(array.get(investArr, ia), 1e18)
            float investB = nz(array.get(investArr, ib), 1e18)
            float deltaA = nz(array.get(riskDeltaPctArr, ia), 1e18)
            float deltaB = nz(array.get(riskDeltaPctArr, ib), 1e18)
            bool doSwap = scoreB < scoreA or (scoreB == scoreA and investB < investA) or (scoreB == scoreA and investB == investA and deltaB < deltaA)
            if doSwap
                int tmp = array.get(displayIdxArr, a)
                array.set(displayIdxArr, a, array.get(displayIdxArr, b))
                array.set(displayIdxArr, b, tmp)
